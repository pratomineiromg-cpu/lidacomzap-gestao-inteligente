import { collection, getDocs } from 'firebase/firestore';
import { db } from '../firebase/firebase';
import { AppRepository, DatabaseState, OpportunityTransitionResult } from './IRepository';
import { ID, Order, OrderItem, Message, TimelineEvent, DispatchQueueEntry, PendingOutboundMessage, FunnelStage, DispatchQueueIdentity, Contact, Conversation, Opportunity, DispatchCampaign, CustomerProfile, CustomerAddress } from '../domain/types';
import { createEmptyDatabaseState } from './firestore/emptyState';
import { COLLECTIONS } from './firestore/schema';
import { fromFirestore } from './firestore/serializers';
import {
  validateContact,
  validateConversation,
  validateMessage,
  validateOpportunity,
  validateDispatchCampaign,
  validateDispatchQueueEntry,
  validatePendingOutboundMessage,
  validateOrder,
  validateOrderItem,
  validateTimelineEvent,
  validateCustomerProfile,
  validateCustomerAddress
} from './firestore/readValidators';

export type HydrationStatus = 'idle' | 'loading' | 'ready' | 'error';

export const OPERATIONAL_COLLECTIONS = [
  COLLECTIONS.CONTACTS,
  COLLECTIONS.CONVERSATIONS,
  COLLECTIONS.MESSAGES,
  COLLECTIONS.OPPORTUNITIES,
  COLLECTIONS.DISPATCH_CAMPAIGNS,
  COLLECTIONS.DISPATCH_QUEUE,
  COLLECTIONS.PENDING_OUTBOUND_MESSAGES,
  COLLECTIONS.ORDERS,
  COLLECTIONS.ORDER_ITEMS,
  COLLECTIONS.TIMELINE_EVENTS,
  COLLECTIONS.CUSTOMER_PROFILES,
  COLLECTIONS.CUSTOMER_ADDRESSES
] as const;

export type AllowedOperationalCollection = typeof OPERATIONAL_COLLECTIONS[number];

export function isOperationalCollectionAllowed(col: string): boolean {
  if (col === COLLECTIONS.USERS) return false;
  return (OPERATIONAL_COLLECTIONS as readonly string[]).includes(col);
}

export class FirestoreReadOnlyRepository implements AppRepository {
  private cache: DatabaseState;
  private status: HydrationStatus = 'idle';
  private errorMessage: string | null = null;
  private generation: number = 0;
  private listeners: Set<() => void> = new Set();
  private firestoreInstance: any;

  constructor(customDb?: any) {
    this.cache = createEmptyDatabaseState();
    this.firestoreInstance = customDb || db;
  }

  // AppReadStore implementation
  getState(): DatabaseState {
    return this.cache;
  }

  subscribe(listener: () => void): () => void {
    this.listeners.add(listener);
    return () => {
      this.listeners.delete(listener);
    };
  }

  private notify(): void {
    this.listeners.forEach(listener => listener());
  }

  // Hydration status getters
  getHydrationStatus(): HydrationStatus {
    return this.status;
  }

  getHydrationError(): string | null {
    return this.errorMessage;
  }

  getHydrationGeneration(): number {
    return this.generation;
  }

  // Write defense in depth: All mutable operations reject before touching Firebase
  async addDispatchQueueEntries(_entries: DispatchQueueEntry[]): Promise<void> {
    throw new Error('Operation is disabled in FIRESTORE_READONLY mode.');
  }

  async prepareDrafts(
    _queueUpdates: { id: ID; status: 'DRAFT_PREPARED'; preparedAt: Date }[],
    _newDrafts: PendingOutboundMessage[]
  ): Promise<ID[]> {
    throw new Error('Operation is disabled in FIRESTORE_READONLY mode.');
  }

  async sendPreparedMessage(
    _message: Message,
    _draftId: ID,
    _queueEntryId: ID,
    _sentAt: Date
  ): Promise<boolean> {
    throw new Error('Operation is disabled in FIRESTORE_READONLY mode.');
  }

  async processClientResponse(
    _message: Message,
    _conversationId: ID,
    _updatedAt: Date,
    _queueEntryToReply?: DispatchQueueIdentity,
    _repliedAt?: Date
  ): Promise<void> {
    throw new Error('Operation is disabled in FIRESTORE_READONLY mode.');
  }

  async createOrderWithItems(
    _order: Order,
    _orderItems: OrderItem[],
    _systemMessage?: Message
  ): Promise<void> {
    throw new Error('Operation is disabled in FIRESTORE_READONLY mode.');
  }

  async updateCampaignDailyLimit(_campaignId: ID, _limit: number, _updatedAt: Date): Promise<void> {
    throw new Error('Operation is disabled in FIRESTORE_READONLY mode.');
  }

  async transitionOpportunityStage(
    _contactId: ID,
    _tenantId: ID,
    _newStage: FunnelStage,
    _timestamp: Date
  ): Promise<OpportunityTransitionResult> {
    throw new Error('Operation is disabled in FIRESTORE_READONLY mode.');
  }

  async addTimelineEvent(_event: TimelineEvent): Promise<void> {
    throw new Error('Operation is disabled in FIRESTORE_READONLY mode.');
  }

  async reset(): Promise<void> {
    throw new Error('Operation is disabled in FIRESTORE_READONLY mode.');
  }

  // Atomic Hydration with stale protection
  async hydrate(customLoader?: (colName: AllowedOperationalCollection) => Promise<any[]>): Promise<boolean> {
    const currentGen = ++this.generation;
    this.status = 'loading';
    this.errorMessage = null;
    this.notify();

    try {
      // 1. Fetch each collection either with customLoader (testable) or getDocs
      const fetchCollectionDocs = async (colName: AllowedOperationalCollection): Promise<any[]> => {
        if (!isOperationalCollectionAllowed(colName)) {
          throw new Error(`Collection ${colName} is not in the operational allowlist.`);
        }

        if (customLoader) {
          return await customLoader(colName);
        }

        const colRef = collection(this.firestoreInstance, colName);
        const snapshot = await getDocs(colRef);
        const docs: any[] = [];
        snapshot.forEach(docSnap => {
          docs.push({
            _docId: docSnap.id,
            ...fromFirestore(docSnap.data())
          });
        });
        return docs;
      };

      // Atomic fetch of all 12 allowed operational collections in parallel
      const [
        rawContacts,
        rawConversations,
        rawMessages,
        rawOpportunities,
        rawCampaigns,
        rawDispatchQueue,
        rawPending,
        rawOrders,
        rawOrderItems,
        rawTimeline,
        rawProfiles,
        rawAddresses
      ] = await Promise.all([
        fetchCollectionDocs(COLLECTIONS.CONTACTS),
        fetchCollectionDocs(COLLECTIONS.CONVERSATIONS),
        fetchCollectionDocs(COLLECTIONS.MESSAGES),
        fetchCollectionDocs(COLLECTIONS.OPPORTUNITIES),
        fetchCollectionDocs(COLLECTIONS.DISPATCH_CAMPAIGNS),
        fetchCollectionDocs(COLLECTIONS.DISPATCH_QUEUE),
        fetchCollectionDocs(COLLECTIONS.PENDING_OUTBOUND_MESSAGES),
        fetchCollectionDocs(COLLECTIONS.ORDERS),
        fetchCollectionDocs(COLLECTIONS.ORDER_ITEMS),
        fetchCollectionDocs(COLLECTIONS.TIMELINE_EVENTS),
        fetchCollectionDocs(COLLECTIONS.CUSTOMER_PROFILES),
        fetchCollectionDocs(COLLECTIONS.CUSTOMER_ADDRESSES)
      ]);

      // 2. Validate & Map strictly into candidate state (throws on any malformed/invalid record)
      const contacts: Contact[] = rawContacts.map(validateContact);
      const conversations: Conversation[] = rawConversations.map(validateConversation);
      const messages: Message[] = rawMessages.map(validateMessage);
      const opportunities: Opportunity[] = rawOpportunities.map(validateOpportunity);
      const campaigns: DispatchCampaign[] = rawCampaigns.map(validateDispatchCampaign);
      const dispatchQueue: DispatchQueueEntry[] = rawDispatchQueue.map(validateDispatchQueueEntry);
      const pendingMessages: PendingOutboundMessage[] = rawPending.map(validatePendingOutboundMessage);
      const orders: Order[] = rawOrders.map(validateOrder);
      const orderItems: OrderItem[] = rawOrderItems.map(validateOrderItem);
      const timeline: TimelineEvent[] = rawTimeline.map(validateTimelineEvent);
      const customerProfiles: CustomerProfile[] = rawProfiles.map(validateCustomerProfile);
      const customerAddresses: CustomerAddress[] = rawAddresses.map(validateCustomerAddress);

      const candidateState: DatabaseState = {
        tenant: {
          id: 't-pratomineiro',
          name: 'Restaurante Prato Mineiro',
          createdAt: new Date(),
          updatedAt: new Date()
        },
        users: [],
        contacts,
        conversations,
        messages,
        pendingMessages,
        opportunities,
        campaigns,
        dispatchQueue,
        orders,
        orderItems,
        timeline,
        products: [],
        customerProfiles,
        customerAddresses
      };

      // 3. Stale guard: Check if this generation is still current
      if (currentGen !== this.generation) {
        return false;
      }

      // 4. Atomic commit: Replace whole cache only after ALL succeeded
      this.cache = candidateState;
      this.status = 'ready';
      this.errorMessage = null;
      this.notify();
      return true;

    } catch (err: any) {
      if (currentGen !== this.generation) {
        return false;
      }

      this.status = 'error';
      this.errorMessage = err?.message || 'Falha ao hidratar dados do Firestore.';
      this.notify();
      return false;
    }
  }
}

export function requireString(value: any, col: string, docId: string, field: string): string {
  if (typeof value !== 'string' || value.trim().length === 0) {
    throw new Error(`Invalid field "${field}" in collection "${col}" doc "${docId}": expected non-empty string.`);
  }
  return value;
}

export function optionalString(value: any, col: string, docId: string, field: string): string | undefined {
  if (value === undefined || value === null) {
    return undefined;
  }
  if (typeof value !== 'string') {
    throw new Error(`Invalid field "${field}" in collection "${col}" doc "${docId}": expected string when provided.`);
  }
  return value.trim().length === 0 ? undefined : value;
}

export function requireFiniteNumber(value: any, col: string, docId: string, field: string): number {
  if (typeof value !== 'number' || !Number.isFinite(value)) {
    throw new Error(`Invalid field "${field}" in collection "${col}" doc "${docId}": expected finite number.`);
  }
  return value;
}

export function parseStrictDate(value: any): Date | null {
  if (value === null || value === undefined) {
    return null;
  }
  if (value instanceof Date) {
    return !Number.isNaN(value.getTime()) ? value : null;
  }
  if (value instanceof Timestamp) {
    const d = value.toDate();
    return !Number.isNaN(d.getTime()) ? d : null;
  }
  if (value && typeof value.toDate === 'function') {
    const d = value.toDate();
    return d instanceof Date && !Number.isNaN(d.getTime()) ? d : null;
  }
  if (value && typeof value.seconds === 'number' && Number.isFinite(value.seconds)) {
    const nanos = typeof value.nanoseconds === 'number' ? value.nanoseconds : 0;
    const d = new Date(value.seconds * 1000 + Math.floor(nanos / 1000000));
    return !Number.isNaN(d.getTime()) ? d : null;
  }
  if (typeof value === 'string' || typeof value === 'number') {
    const d = new Date(value);
    return !Number.isNaN(d.getTime()) ? d : null;
  }
  return null;
}

export function requireDate(value: any, col: string, docId: string, field: string): Date {
  const d = parseStrictDate(value);
  if (!d) {
    throw new Error(`Invalid field "${field}" in collection "${col}" doc "${docId}": expected valid date/timestamp.`);
  }
  return d;
}

export function optionalDate(value: any, col: string, docId: string, field: string): Date | undefined {
  if (value === undefined || value === null) {
    return undefined;
  }
  const d = parseStrictDate(value);
  if (!d) {
    throw new Error(`Invalid field "${field}" in collection "${col}" doc "${docId}": expected valid date/timestamp when provided.`);
  }
  return d;
}

export function requireStringArray(value: any, col: string, docId: string, field: string): string[] {
  if (!Array.isArray(value) || value.some(item => typeof item !== 'string')) {
    throw new Error(`Invalid field "${field}" in collection "${col}" doc "${docId}": expected array of strings.`);
  }
  return value;
}

export function requireEnum<T extends string>(value: any, allowed: readonly T[], col: string, docId: string, field: string): T {
  if (typeof value !== 'string' || !(allowed as readonly string[]).includes(value)) {
    throw new Error(`Invalid enum "${field}" in collection "${col}" doc "${docId}": expected one of [${allowed.join(', ')}].`);
  }
  return value as T;
}

    neighborhood: requireString(d?.neighborhood, 'customerAddresses', docId, 'neighborhood'),
    complement: optionalString(d?.complement, 'customerAddresses', docId, 'complement'),

import { AppRepository } from './IRepository';
import { mockRepository } from './MockRepository';
import { FirestoreReadOnlyRepository } from './FirestoreReadOnlyRepository';
import { getDataSourceMode, DataSourceMode } from '../config/dataSource';

export function createRepository(modeOverride?: DataSourceMode): AppRepository {
  const mode = modeOverride || getDataSourceMode();
  switch (mode) {
    case 'MOCK':
      return mockRepository;
    case 'FIRESTORE_READONLY':
      return new FirestoreReadOnlyRepository();
    case 'FIRESTORE':
      throw new Error('Firestore write mode is not enabled.');
    default:
      throw new Error(`Unsupported data source mode: ${mode}`);
  }
}

// Default export uses the configured data source mode
export const repository: AppRepository = createRepository();

import React, { useEffect, useState, useSyncExternalStore } from 'react';
import { getDataSourceMode } from '../config/dataSource';
import { repository } from '../repositories';
import { FirestoreReadOnlyRepository, HydrationStatus } from '../repositories/FirestoreReadOnlyRepository';

export const DataSourceGate: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const mode = getDataSourceMode();

  if (mode === 'MOCK') {
    return <>{children}</>;
  }

  if (mode === 'FIRESTORE') {
    return (
      <div className="min-h-screen flex items-center justify-center bg-slate-50 p-6">
        <div className="max-w-md bg-white p-8 rounded-xl border border-red-200 shadow-sm text-center">
          <h2 className="text-lg font-bold text-red-600 mb-2">Modo não permitido</h2>
          <p className="text-sm text-slate-600">Firestore write mode is not enabled.</p>
        </div>
      </div>
    );
  }

  // FIRESTORE_READONLY Mode
  const readOnlyRepo = repository as FirestoreReadOnlyRepository;

  const hydrationStatus: HydrationStatus = useSyncExternalStore(
    readOnlyRepo.subscribe ? readOnlyRepo.subscribe.bind(readOnlyRepo) : () => () => {},
    () => (typeof readOnlyRepo.getHydrationStatus === 'function' ? readOnlyRepo.getHydrationStatus() : 'ready')
  );

  useEffect(() => {
    if (typeof readOnlyRepo.getHydrationStatus === 'function') {
      if (readOnlyRepo.getHydrationStatus() === 'idle') {
        readOnlyRepo.hydrate();
      }
    }
  }, [readOnlyRepo]);

  if (hydrationStatus === 'idle' || hydrationStatus === 'loading') {
    return (
      <div className="min-h-screen flex flex-col items-center justify-center bg-slate-50 text-slate-600 p-6">
        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-indigo-600 mb-4" />
        <p className="text-sm font-medium text-slate-700">Carregando dados operacionais...</p>
      </div>
    );
  }

  if (hydrationStatus === 'error') {
    return (
      <div className="min-h-screen flex flex-col items-center justify-center bg-slate-50 p-6 text-center">
        <div className="max-w-md bg-white p-8 rounded-xl border border-slate-200 shadow-sm">
          <div className="w-12 h-12 bg-red-50 text-red-600 rounded-full flex items-center justify-center mx-auto mb-4 font-bold text-xl">
            !
          </div>
          <h2 className="text-lg font-bold text-slate-800 mb-2">Não foi possível carregar os dados operacionais.</h2>
          <p className="text-sm text-slate-500 mb-6">
            Ocorreu uma falha ao sincronizar com o banco de dados em modo somente leitura.
          </p>
          <button
            onClick={() => readOnlyRepo.hydrate()}
            className="px-5 py-2.5 bg-indigo-600 hover:bg-indigo-700 text-white text-sm font-medium rounded-lg shadow-sm transition-colors cursor-pointer"
          >
            Tentar novamente
          </button>
        </div>
      </div>
    );
  }

  return <>{children}</>;
};

import React from 'react';
import { BrowserRouter, Routes, Route } from 'react-router-dom';
import { Layout } from './components/Layout';
import { Dashboard } from './views/Dashboard';
import { Central } from './views/Central';
import { CRM } from './views/CRM';

import { Pedidos } from './views/Pedidos';
import { AuthProvider, useAuth } from './auth/AuthProvider';
import { LoginView } from './views/LoginView';
import { UnauthorizedView } from './views/UnauthorizedView';
import { DataSourceGate } from './components/DataSourceGate';

const AuthGate: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const { status } = useAuth();

  if (status === 'loading' || status === 'authorizing') {
    return (
      <div className="min-h-screen flex items-center justify-center bg-gray-50">
        <div className="text-gray-500">Carregando...</div>
      </div>
    );
  }

  if (status === 'unauthenticated') {
    return <LoginView />;
  }

  if (status === 'unauthorized') {
    return <UnauthorizedView />;
  }

  if (status === 'authorized') {
    return <>{children}</>;
  }

  return null;
};

  return (
    <AuthProvider>
      <AuthGate>
        <DataSourceGate>

                <Route index element={<Dashboard />} />
                <Route path="chat" element={<Central />} />
                <Route path="crm" element={<CRM />} />
                <Route path="disparador" element={<Disparador />} />
                <Route path="pedidos" element={<Pedidos />} />
              </Route>
            </Routes>
          </BrowserRouter>
        </DataSourceGate>
      </AuthGate>
    </AuthProvider>
  );
}
